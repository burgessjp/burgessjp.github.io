# MBTI 测试单文件 Web 应用 — 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 交付一个零依赖单 HTML 文件的 MBTI 测试（60 题、四维度计分、16 类型结果页、canvas 分享卡片），断网双击即用，移动端优先。

**Architecture:** 单 `index.html`，三个 script 块按标记分区：`SCORING-CORE`（纯计分逻辑，无 DOM）、`CONTENT-DATA`（QUESTIONS 60 题 + TYPES 16 类型）、匿名 app 块（DOM/状态/分享卡片）。三屏状态机 `welcome → quiz → result`，进度持久化 localStorage。

**Tech Stack:** vanilla JS + 内联 CSS，无任何外部资源；Node 断言脚本（`test/scoring.test.mjs`，零依赖）；E2E 用 webapp-testing（Playwright）。

**Spec:** `docs/superpowers/specs/2026-11-13-mbti-test-design.md`

## Global Constraints

- `index.html` 禁止引用任何外部资源：无 CDN、无外链字体/图片、无网络请求
- 禁止 fonts.googleapis.com；字体只用系统栈：`-apple-system, "PingFang SC", "Microsoft YaHei", sans-serif`
- 文案简体中文口语化；代码、变量、文件名英文；变量 camelCase，常量 UPPER_SNAKE_CASE
- 每维度 15 题，其中 `dir:+1` 恰好 8 题、`dir:-1` 恰好 7 题
- localStorage key `mbti-test-state`，结构 `{v:1, phase, order, answers, index, result}`，访问全部 try/catch
- UI 元素挂稳定 `data-testid`，视觉打磨（Task 7）不得改动或删除它们
- 提交信息用 Conventional Commits，禁止 Co-Authored-By
- 每个任务收尾必须跑 `node test/scoring.test.mjs` 全绿后再 commit

```js
import { readFileSync } from 'node:fs';

const html = readFileSync(new URL('../index.html', import.meta.url), 'utf8');

function loadBlock(marker, names) {
  const re = new RegExp(`<!--\\s*${marker}-START\\s*-->([\\s\\S]*?)<!--\\s*${marker}-END\\s*-->`);
  const m = html.match(re);
  if (!m) throw new Error(`marker ${marker} not found in index.html`);
  const inner = m[1].replace(/<script[^>]*>/g, '').replace(/<\/script>/g, '');
  const fn = new Function(`${inner}\n; return { ${names.join(',')} };`);
  return fn();
}

let passed = 0;
function check(name, cond, detail = '') {
  if (!cond) { console.error(`FAIL: ${name}${detail ? ' — ' + detail : ''}`); process.exit(1); }
  passed++;
}

// ── SCORING-CORE 接口存在性 ──
const { DIMENSIONS, DIM_LABELS, LIKERT, computeResult, shuffle } =
  loadBlock('SCORING-CORE', ['DIMENSIONS', 'DIM_LABELS', 'LIKERT', 'computeResult', 'shuffle']);

check('DIMENSIONS 四维度', DIMENSIONS.length === 4 && DIMENSIONS.join('') === 'EISNTFJP');
check('DIM_LABELS 两极字母', DIM_LABELS.EI[0] === 'E' && DIM_LABELS.EI[1] === 'I');
check('LIKERT 五点', LIKERT.length === 5 && LIKERT[0].v === -2 && LIKERT[4].v === 2);

// ── 计分数学（手算 fixture，不依赖真实题库）──
const FX = [
  { id: 'f1', dim: 'EI', dir:  1, text: 'fx1' },
  { id: 'f2', dim: 'EI', dir: -1, text: 'fx2' },
  { id: 'f3', dim: 'SN', dir:  1, text: 'fx3' },
];
// EI: +2*1 + +1*(-1) = +1 → E, max=4, pct=50+1/4*50=62.5→63
// SN: -1*1 = -1 → N, max=2, pct=50+1/2*50=75
// TF/JP 无题 → raw 0 → tie-break 第二极, pct 50 → type = ENFP
const r1 = computeResult(FX, { f1: 2, f2: 1, f3: -1 });
check('fixture 类型 ENFP', r1.type === 'ENFP', `got ${r1.type}`);
check('fixture EI', r1.dims.EI.pole === 'E' && r1.dims.EI.pct === 63 && r1.dims.EI.raw === 1);
check('fixture SN', r1.dims.SN.pole === 'N' && r1.dims.SN.pct === 75 && r1.dims.SN.raw === -1);
check('fixture TF tie-break', r1.dims.TF.pole === 'F' && r1.dims.TF.pct === 50 && r1.dims.TF.raw === 0);
check('fixture JP tie-break', r1.dims.JP.pole === 'P' && r1.dims.JP.pct === 50);

// 答案钳制与缺省：越界值按最近合法值计，缺失按 0 计
const r2 = computeResult(FX, { f1: 9, f2: -9 });
check('钳制/缺省', r2.dims.EI.raw === 2 && r2.dims.SN.raw === 0, `got EI=${r2.dims.EI.raw} SN=${r2.dims.SN.raw}`);

// 确定性：同输入两次调用结果一致
check('确定性', JSON.stringify(computeResult(FX, { f1: 2 })) === JSON.stringify(computeResult(FX, { f1: 2 })));

// shuffle：不改入参、元素守恒
const arr = [1, 2, 3, 4, 5, 6, 7, 8];
const before = arr.join(',');
const sh = shuffle(arr);
check('shuffle 守恒', arr.join(',') === before && sh.slice().sort().join(',') === '1,2,3,4,5,6,7,8');

console.log(`OK: ${passed} checks passed`);
```

```html
<!-- SCORING-CORE-START -->
<script>
const DIMENSIONS = ['EI', 'SN', 'TF', 'JP'];
const DIM_LABELS = { EI: ['E', 'I'], SN: ['S', 'N'], TF: ['T', 'F'], JP: ['J', 'P'] };
const LIKERT = [
  { v: -2, label: '非常不同意' },
  { v: -1, label: '不同意' },
  { v:  0, label: '中立' },
  { v:  1, label: '同意' },
  { v:  2, label: '非常同意' },
];

function computeResult(questions, answers) {
  const dims = {};
  for (const dim of DIMENSIONS) {
    let raw = 0;
    let max = 0;
    for (const q of questions) {
      if (q.dim !== dim) continue;
      const a = Math.max(-2, Math.min(2, answers[q.id] ?? 0));
      raw += a * q.dir;
      max += 2;
    }
    const [first, second] = DIM_LABELS[dim];
    const pole = raw > 0 ? first : second;   // tie-break：0 归第二极
    const pct = Math.round(50 + (max > 0 ? Math.abs(raw) / max * 50 : 0));
    dims[dim] = { pole, pct, raw };
  }
  const type = DIMENSIONS.map(d => dims[d].pole).join('');
  return { type, dims };
}

function shuffle(arr) {
  const a = arr.slice();
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}
</script>
<!-- SCORING-CORE-END -->
</body>
</html>
```

- [x] **Step 3: 撰写 16 类型档案**

**撰写规则（硬性）：**

1. `name`：中文称号（2–6 字），用流传最广的版本，如 INTP「逻辑学家」、ENFJ「主人公」、ISTP「鉴赏家」
2. `tagline`：分享卡片上的一句话，≤15 字，要有传播欲，如 INTP「安静的大脑，轰鸣的宇宙」
3. `desc`：150–250 字。结构：这个类型怎么看世界（1–2 句）→ 行事风格（2–3 句）→ 别人眼里常是什么样的（1 句）。口吻温暖但诚实，不捧不贬
4. `strengths` 3–4 条、`weaknesses` 恰 3 条：每条一句话（≥6 字），缺点写真实但不刻薄的（如「想太多，行动慢半拍」）
5. `careers` 4–6 个：职业方向词组，不写具体公司
6. `famous` 3–4 位：公开流传的常见归类，可用虚构角色补充；写「名字 · 领域」或纯名字
7. `relationship`：80–150 字，和朋友/恋人相处这个类型的注意事项，用「你」称呼读者
8. 全部原创中文，基于类型学公开共识；每写完 4 个类型回头读一遍，确保 16 个之间语气、颗粒度一致、无复制粘贴句式

```html
(function () {
  'use strict';
  var STORE_KEY = 'mbti-test-state';
  var app = document.getElementById('app');

  function newState() {
    return { v: 1, phase: 'welcome', order: shuffle(QUESTIONS.map(function (q) { return q.id; })),
             answers: {}, index: 0, result: null };

  function loadState() {
    try {
      var raw = localStorage.getItem(STORE_KEY);
      if (!raw) return null;
      var s = JSON.parse(raw);
      if (s.v !== 1 || !Array.isArray(s.order) || s.order.length !== QUESTIONS.length) return null;
      return s;

  function save() { try { localStorage.setItem(STORE_KEY, JSON.stringify(state)); } catch (e) {} }
  function clearStore() { try { localStorage.removeItem(STORE_KEY); } catch (e) {} }

  var state = loadState() || newState();

  function el(tag, attrs, children) {
    var node = document.createElement(tag);
    for (var k in attrs) {
      if (attrs[k] == null) continue;
      if (k === 'text') node.textContent = attrs[k];
      else if (k === 'html') node.innerHTML = attrs[k];
      else if (k.slice(0, 2) === 'on') node.addEventListener(k.slice(2), attrs[k]);
      else node.setAttribute(k, attrs[k]);
    }
    (children || []).forEach(function (c) { node.appendChild(c); });

    screen.appendChild(el('h3', { text: '同款名人' }));
    screen.appendChild(el('p', { text: p.famous.join(' · ') }));
    screen.appendChild(el('h3', { text: '和 ' + p.name + ' 相处' }));
    screen.appendChild(el('p', { 'class': 'desc', text: p.relationship }));

    screen.appendChild(el('button', { 'data-testid': 'btn-share', 'class': 'btn-primary', disabled: 'disabled', text: '生成分享卡片' }));

})();
```

```js
  var CARD_STYLE = { bg: '#1c1917', fg: '#ffffff', accent: '#f59e0b', muted: '#a8a29e', w: 750, h: 1000 };

  function drawShareCard(r) {
    var s = CARD_STYLE;
    var c = document.createElement('canvas');
    c.width = s.w; c.height = s.h;
    var ctx = c.getContext('2d');
    var FONT = '-apple-system, "PingFang SC", "Microsoft YaHei", sans-serif';
    ctx.fillStyle = s.bg; ctx.fillRect(0, 0, s.w, s.h);
    ctx.fillStyle = s.fg; ctx.textAlign = 'center';
    ctx.font = '600 26px ' + FONT;
    ctx.fillText('MBTI 性格测试', s.w / 2, 90);
    ctx.font = '800 170px ' + FONT;
    ctx.fillStyle = s.accent;
    ctx.fillText(r.type, s.w / 2, 300);
    ctx.fillStyle = s.fg;
    ctx.font = '600 56px ' + FONT;
    ctx.fillText(TYPES[r.type].name, s.w / 2, 395);
    ctx.font = '400 34px ' + FONT;
    ctx.fillStyle = s.muted;
    ctx.fillText(TYPES[r.type].tagline, s.w / 2, 455);

    var y = 560;
    DIMENSIONS.forEach(function (dim) {
      var d = r.dims[dim];
      var first = DIM_LABELS[dim][0], second = DIM_LABELS[dim][1];
      var leftPct = d.pole === first ? d.pct : 100 - d.pct;
      var x0 = 110, x1 = s.w - 110;
      ctx.textAlign = 'right'; ctx.fillStyle = d.pole === first ? s.accent : s.muted;
      ctx.font = '700 44px ' + FONT; ctx.fillText(first, x0 - 24, y + 16);
      ctx.textAlign = 'left'; ctx.fillStyle = d.pole === second ? s.accent : s.muted;
      ctx.fillText(second, x1 + 24, y + 16);
      ctx.fillStyle = 'rgba(255,255,255,.14)'; ctx.fillRect(x0, y, x1 - x0, 18);
      ctx.fillStyle = s.accent;
      if (d.pole === first) ctx.fillRect(x1 - (x1 - x0) * leftPct / 100, y, (x1 - x0) * leftPct / 100, 18);
      else ctx.fillRect(x0, y, (x1 - x0) * leftPct / 100, 18);
      ctx.textAlign = 'center'; ctx.fillStyle = s.fg; ctx.font = '600 30px ' + FONT;
      ctx.fillText(d.pole + ' ' + d.pct + '%', s.w / 2, y + 62);
      y += 110;
    });

    ctx.textAlign = 'center'; ctx.fillStyle = s.muted; ctx.font = '400 26px ' + FONT;
    ctx.fillText('长按保存 · 发给朋友看看准不准', s.w / 2, s.h - 60);
    return c.toDataURL('image/png');
  }

  function shareCard() {
    var dataUrl = drawShareCard(state.result);
    var btn = document.querySelector('[data-testid="btn-share"]');
    var file = null;
    try { file = new File([dataUrlToBlob(dataUrl)], 'mbti-' + state.result.type + '.png', { type: 'image/png' }); } catch (e) {}
    if (file && navigator.canShare && navigator.canShare({ files: [file] })) {
      navigator.share({ files: [file], title: '我的 MBTI 是 ' + state.result.type }).catch(function () {});
      return;
    }
    var wrap = el('div', { 'class': 'share-wrap' });
    wrap.appendChild(el('img', { 'data-testid': 'share-img', 'class': 'share-img', src: dataUrl, alt: 'MBTI 分享卡片' }));
    wrap.appendChild(el('a', { 'data-testid': 'share-download', 'class': 'btn-primary', download: 'mbti-' + state.result.type + '.png',
      href: dataUrl, text: '下载卡片' }));
    btn.replaceWith(wrap);
  }
  function dataUrlToBlob(dataUrl) {
    var parts = dataUrl.split(',');
    var bin = atob(parts[1]);
    var arr = new Uint8Array(bin.length);
    for (var i = 0; i < bin.length; i++) arr[i] = bin.charCodeAt(i);
    return new Blob([arr], { type: 'image/png' });
  }
```
